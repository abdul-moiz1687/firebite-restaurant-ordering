const STATUS_CONFIG = {
  pending: { label: "Pending", color: "#F59E0B" },
  confirmed: { label: "Confirmed", color: "#3B82F6" },
  preparing: { label: "Preparing", color: "#8B5CF6" },

function formatCurrency(amount, currency = "PKR") {
  return new Intl.NumberFormat("en-PK", {
    style: "currency",
    currency,
    maximumFractionDigits: 0,
  }).format(Number(amount) || 0);
}

function formatCompactCurrency(amount) {
  return `₨${new Intl.NumberFormat("en-PK", {
    notation: "compact",
    maximumFractionDigits: 1,
  }).format(Number(amount) || 0)}`;
}

function RevenueChart({ data = [] }) {
  const chartWidth = 720;
  const left = 82;
  const right = 24;
  const top = 22;
  const bottom = 202;
  const plotWidth = chartWidth - left - right;
  const plotHeight = bottom - top;
  const values = data.map((item) => Number(item.amount) || 0);
  const maximum = Math.max(...values, 1);

  const points = data.map((item, index) => {
    const x =
      data.length <= 1
        ? left + plotWidth / 2
        : left + (index * plotWidth) / (data.length - 1);
    const y = bottom - ((Number(item.amount) || 0) / maximum) * plotHeight;

    return { x, y, ...item };
  });

  const linePath = points
    .map((point, index) => `${index === 0 ? "M" : "L"} ${point.x} ${point.y}`)
    .join(" ");

  const areaPath = points.length
    ? `${linePath} L ${points[points.length - 1].x} ${bottom} L ${points[0].x} ${bottom} Z`
    : "";

  const hasRevenue = values.some((value) => value > 0);

  return (
    <div className="w-full">
      <div className="mb-5 flex flex-wrap items-center justify-between gap-3">
        <div>
          <p className="text-sm font-semibold text-slate-800">
            Delivered order value
          </p>
          <p className="mt-1 text-xs text-slate-500">
            Daily value of orders created in the last 7 days and marked Delivered
          </p>
        </div>
        <span className="inline-flex items-center gap-2 rounded-full bg-orange-50 px-3 py-1.5 text-xs font-semibold text-orange-700">
          <CalendarDays size={14} />
          Last 7 days
        </span>
      </div>

      {data.length === 0 ? (
        <div className="flex h-64 items-center justify-center text-sm text-slate-400">
          Revenue data is not available yet.
        </div>
      ) : (
        <div className="w-full overflow-hidden">
          <svg
            viewBox="0 0 720 250"
            role="img"
            aria-label="Delivered order value over the last seven days"
            className="h-auto w-full overflow-visible"
          >
            <defs>
              <linearGradient id="firebite-revenue-fill" x1="0" x2="0" y1="0" y2="1">
                <stop offset="0%" stopColor="#F97316" stopOpacity="0.22" />
                <stop offset="100%" stopColor="#F97316" stopOpacity="0.015" />
              </linearGradient>
            </defs>

            {[0, 1, 2, 3].map((line) => {
              const y = top + (plotHeight / 3) * line;
              const value = maximum * ((3 - line) / 3);

              return (
                <g key={line}>
                  <line
                    x1={left}
                    y1={y}
                    x2={chartWidth - right}
                    y2={y}
                    stroke="#E8ECF2"
                    strokeDasharray="4 6"
                  />
                  <text
                    x={left - 12}
                    y={y + 4}
                    textAnchor="end"
                    fill="#94A3B8"
                    fontSize="11"
                  >
                    {formatCompactCurrency(value)}
                  </text>
                </g>
              );
            })}

            {hasRevenue && (
              <path d={areaPath} fill="url(#firebite-revenue-fill)" />
            )}

            {hasRevenue && (
              <path
                d={linePath}
                fill="none"
                stroke="#F97316"
                strokeWidth="3.5"
                strokeLinecap="round"
                strokeLinejoin="round"
              />
            )}

            {points.map((point) => (
              <g key={point.date}>
                <circle
                  cx={point.x}
                  cy={point.y}
                  r="5"
                  fill="#FFFFFF"
                  stroke="#F97316"
                  strokeWidth="3"
                >
                  <title>
                    {`${point.date}: ${formatCurrency(point.amount)}`}
                  </title>
                </circle>
                <text
                  x={point.x}
                  y="232"
                  textAnchor="middle"
                  fill="#64748B"
                  fontSize="11"
                  fontWeight="600"
                >
                  {point.label}
                </text>
              </g>
            ))}

            {!hasRevenue && (
              <text
                x={left + plotWidth / 2}
                y={top + plotHeight / 2}
                textAnchor="middle"
                fill="#94A3B8"
                fontSize="13"
              >
                No delivered sales recorded in this period
              </text>
            )}
          </svg>
        </div>
      )}
      <p className="mt-2 text-[11px] leading-5 text-slate-400">
        This figure uses the subtotal of orders marked Delivered. Payment collection is not separately recorded in the current order data.
      </p>
    </div>
  );
}

function OrderStatusChart({ items = [] }) {
  const total = items.reduce((sum, item) => sum + (Number(item.count) || 0), 0);
  const radius = 54;
  const circumference = 2 * Math.PI * radius;
  let offset = 0;

  const segments = items.map((item) => {
    const count = Number(item.count) || 0;
    const length = total > 0 ? (count / total) * circumference : 0;
    const segment = {
      ...item,
      length,
      offset,
      color: STATUS_CONFIG[item.status]?.color || "#CBD5E1",
      label: STATUS_CONFIG[item.status]?.label || item.status,
    };
    offset += length;
    return segment;
  });

  return (
    <div className="flex flex-col items-center gap-7 sm:flex-row sm:justify-around">
      <div className="relative h-44 w-44 shrink-0">
        <svg viewBox="0 0 160 160" role="img" aria-label="Order status distribution" className="h-full w-full">
          <circle
            cx="80"
            cy="80"
            r={radius}
            fill="none"
            stroke="#EEF1F5"
            strokeWidth="15"
          />
          {segments.map((segment) =>
            segment.length > 0 ? (
              <circle
                key={segment.status}
                cx="80"
                cy="80"
                r={radius}
                fill="none"
                stroke={segment.color}
                strokeWidth="15"
                strokeDasharray={`${segment.length} ${circumference - segment.length}`}
                strokeDashoffset={-segment.offset}
                transform="rotate(-90 80 80)"
                className="transition-all duration-700 ease-out"
              >
                <title>{`${segment.label}: ${segment.count}`}</title>
              </circle>
            ) : null
          )}
        </svg>
        <div className="pointer-events-none absolute inset-0 flex flex-col items-center justify-center">
          <span className="text-3xl font-black tracking-tight text-slate-900">
            {total}
          </span>
          <span className="mt-0.5 text-[11px] font-semibold uppercase tracking-wider text-slate-400">
            Total orders
          </span>
        </div>
      </div>

      <div className="grid w-full grid-cols-1 gap-3 sm:max-w-[230px]">
        {segments.map((segment) => (
          <div key={segment.status} className="flex items-center justify-between gap-3">
            <div className="flex min-w-0 items-center gap-2.5">
              <span
                className="h-2.5 w-2.5 shrink-0 rounded-full"
                style={{ backgroundColor: segment.color }}
              />
              <span className="truncate text-xs font-medium text-slate-600">
                {segment.label}
              </span>
            </div>
            <span className="text-xs font-bold tabular-nums text-slate-900">
              {segment.count}
            </span>
          </div>
        ))}
        {total === 0 && (
          <p className="text-xs text-slate-400">
            Status distribution will appear when orders arrive.
          </p>
        )}
      </div>
    </div>
  );
}

const DAY_MS = 24 * 60 * 60 * 1000;
const PAKISTAN_OFFSET_MS = 5 * 60 * 60 * 1000;

function getPakistanDateKey(date) {
  const parts = new Intl.DateTimeFormat("en-CA", {
    timeZone: "Asia/Karachi",
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
  }).formatToParts(date);
  const values = Object.fromEntries(
    parts.map(({ type, value }) => [type, value])
  );

  return `${values.year}-${values.month}-${values.day}`;
}

function getRevenueDays(now) {
  const todayKey = getPakistanDateKey(now);
  const [year, month, day] = todayKey.split("-").map(Number);
  const pakistanMidnightUtc =
    Date.UTC(year, month - 1, day) - PAKISTAN_OFFSET_MS;
  const firstDayUtc = pakistanMidnightUtc - 6 * DAY_MS;

  return Array.from({ length: 7 }, (_, index) => {
    const date = new Date(firstDayUtc + index * DAY_MS);

    return {
      date: getPakistanDateKey(date),
      label: new Intl.DateTimeFormat("en-PK", {
        timeZone: "Asia/Karachi",
        weekday: "short",
      }).format(date),
      amount: 0,
    };
  });
}

      deliveredValueSnapshot,
      weekOrdersSnapshot,
    ] = await Promise.all([
      ordersRef.count().get(),
      ordersRef
        .where("createdAt", ">=", Timestamp.fromDate(todayStart))
        .count()
        .get(),
      Promise.all(
        ORDER_STATUSES.map((status) =>
          ordersRef.where("status", "==", status).count().get()
        )
      ),
      ordersRef
        .where("status", "==", "delivered")
        .aggregate({
          deliveredValue: AggregateField.sum("subtotal"),
        })
        .get(),
      ordersRef
        .where("createdAt", ">=", Timestamp.fromDate(weekStart))
        .get(),
    ]);

    const statusBreakdown = ORDER_STATUSES.map((status, index) => ({
      status,
      count: statusCountSnapshots[index].data().count,
    }));

    for (const document of weekOrdersSnapshot.docs) {
      const order = document.data();

      if (order.status !== "delivered") continue;

      const createdAt = order.createdAt?.toDate?.();
      const amount = Number(order.subtotal);

      if (!(createdAt instanceof Date) || !Number.isFinite(amount)) {
        continue;
      }

      const day = revenueByDate.get(getPakistanDateKey(createdAt));

      if (day) {
        day.amount += amount;
      }
    }

    const countFor = (status) =>
      statusBreakdown.find((item) => item.status === status)?.count ?? 0;

    return Response.json(
      {
        statistics: {
          totalOrders: totalOrdersSnapshot.data().count,
          todayOrders: todayOrdersSnapshot.data().count,
          pendingOrders: countFor("pending"),
          deliveredOrders: countFor("delivered"),
          deliveredOrderValue:
            deliveredValueSnapshot.data().deliveredValue ?? 0,
        },
        statusBreakdown,
        revenueTrend: revenueDays,
      },
      {
        headers: { "Cache-Control": "no-store, max-age=0" },
      }
    );
  } catch (error) {
    console.error("Admin dashboard API error:", error);

    return Response.json(
      { error: "Unable to load dashboard data." },
      {
        status: 500,
        headers: { "Cache-Control": "no-store, max-age=0" },
      }
    );
  }
}
